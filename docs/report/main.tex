\section{Khai thác và Đặc tả Yêu cầu (Task 1)}

\subsection{Bối cảnh nghiệp vụ, Stakeholder và Phạm vi dự án}
\textit{(Phụ trách thực hiện: Phan Lê Ngọc Như Ý)}

\subsubsection{Bối cảnh nghiệp vụ}
BrewLite là hệ thống đặt món và quản lý hoạt động của chuỗi cửa hàng đồ uống. Hệ thống hướng đến việc chuẩn hóa quy trình từ xem thực đơn, lựa chọn sản phẩm, đặt món, thanh toán không tiền mặt đến theo dõi trạng thái đơn hàng trên một nền tảng web tối ưu cho thiết bị di động (Mobile-first/PWA). Đồng thời, hệ thống hỗ trợ cửa hàng quản lý tập trung thông tin sản phẩm, tồn kho, đơn hàng, khuyến mãi và giao dịch điểm thưởng.

Trong quy trình rời rạc, khách hàng có thể phải tiếp cận thực đơn qua nhiều kênh khác nhau, khó nắm bắt tình trạng còn hàng và trạng thái xử lý đơn. Nhân viên tiếp nhận yêu cầu thủ công có thể nhầm lẫn về sản phẩm, số lượng, Size, Topping hoặc ghi chú. Việc quản lý giá bán, tồn kho, thanh toán và điểm thưởng ở nhiều nơi cũng làm tăng nguy cơ sai lệch dữ liệu và khó đối soát.

BrewLite giải quyết các vấn đề trên bằng một luồng nghiệp vụ thống nhất. Khách hàng có thể duyệt thực đơn, tìm kiếm và lọc sản phẩm, xem thông tin chi tiết, lựa chọn Size/Topping, quản lý giỏ hàng, xác nhận đơn, thanh toán và theo dõi trạng thái. Các nghiệp vụ quan trọng như kiểm tra tồn kho, tính tổng tiền, áp dụng khuyến mãi và ghi nhận điểm thưởng được xử lý tại backend nhằm hạn chế sai lệch giữa giao diện và dữ liệu hệ thống.

Các mục tiêu nghiệp vụ chính gồm:
\begin{enumerate}
    \item Cung cấp thực đơn tập trung, hỗ trợ tìm kiếm, lọc danh mục và xem chi tiết sản phẩm.
    \item Hạn chế sai sót khi khách hàng lựa chọn sản phẩm, Size, Topping và số lượng.
    \item Chuẩn hóa quy trình tạo đơn, kiểm tra tồn kho và tính tiền tại máy chủ.
    \item Hỗ trợ thanh toán không tiền mặt, khuyến mãi và tích điểm có kiểm soát.
    \item Minh bạch trạng thái đơn hàng để khách hàng và nhân viên cùng theo dõi.
    \item Xây dựng nền tảng có thể kiểm thử, triển khai và mở rộng theo kiến trúc của dự án.
\end{enumerate}

\subsubsection{Phạm vi dự án (In-scope \& Out-of-scope)}
\begin{itemize}
    \item \textbf{Trong phạm vi (In-scope):}
    \begin{itemize}
        \item \textit{Thực đơn và sản phẩm:} danh sách sản phẩm, tìm kiếm, lọc theo danh mục, phân trang, xem chi tiết, giá, mô tả, hình ảnh và tình trạng còn hàng; lựa chọn Size/Topping trên giao diện.
        \item \textit{Tài khoản:} đăng ký, đăng nhập, xác thực JWT và xem hồ sơ người dùng.
        \item \textit{Giỏ hàng và đặt món:} thêm, chỉnh sửa, xóa món; thay đổi số lượng và tùy chọn; xác nhận đơn; kiểm tra tồn kho và tính tổng tiền ở backend.
        \item \textit{Thanh toán và ưu đãi:} thanh toán Ví/Thẻ/QR mô phỏng, khóa chống trùng lặp \texttt{Idempotency-Key}, áp dụng mã khuyến mãi (\texttt{SGU2026}, \texttt{BREWLITE10}, \texttt{FREESHIP}) và ghi nhận điểm thưởng Loyalty.
        \item \textit{Theo dõi đơn hàng:} xem xác nhận đơn vừa đặt, lịch sử đơn (\texttt{GET /orders/me}) và tiến trình xử lý đơn.
        \item \textit{Dữ liệu và vận hành:} lưu trữ 7 thực thể trong PostgreSQL qua Prisma; endpoint kiểm tra sức khỏe \texttt{GET /health}; kiểm thử tự động 3 tầng và triển khai CI/CD.
    \end{itemize}
    \item \textbf{Ngoài phạm vi (Out-of-scope):}
    \begin{itemize}
        \item Ứng dụng native riêng cho iOS/Android; MVP tập trung vào web responsive theo định hướng PWA.
        \item Tích hợp thiết bị POS, máy in hóa đơn, máy quét mã vạch hoặc phần cứng chuyên dụng tại quầy.
        \item Quản lý mua hàng, nhà cung cấp, công nợ, kế toán tổng hợp, nhân sự và tính lương.
        \item Đặt bàn, quản lý chỗ ngồi, điều phối giao hàng hoặc theo dõi lộ trình shipper.
        \item Kết nối trực tiếp cổng ngân hàng thương mại thực tế (sử dụng dịch vụ thanh toán sandbox/mô phỏng).
        \item Dự báo nhu cầu hay gợi ý món bằng AI.
    \end{itemize}
\end{itemize}

\subsubsection{Giả định, ràng buộc và rủi ro}
\begin{enumerate}
    \item Giá sản phẩm được lưu dưới dạng số nguyên theo đơn vị đồng Việt Nam (VND).
    \item Dữ liệu sản phẩm và mã khuyến mãi mẫu được khởi tạo thông qua cơ chế Database Seed chạy lặp an toàn.
    \item Phụ thu Size/Topping được chốt thống nhất và tính toán lại ở backend khi tạo đơn hàng (\texttt{POST /orders}) để đảm bảo tính toàn vẹn dữ liệu.
    \item MVP sử dụng số lượng tồn kho chung (\texttt{stock}) cho từng sản phẩm trên toàn hệ thống.
    \item PWA yêu cầu chạy trên HTTPS (Production) hoặc \texttt{localhost} (môi trường phát triển) để kích hoạt Service Worker.
    \item Hệ thống sử dụng Next.js, NestJS, PostgreSQL và Prisma trong cấu trúc Monorepo; các thành viên tích hợp thông qua Git branch và Pull Request.
\end{enumerate}

\caption{Sơ đồ Use-case chi tiết cho nghiệp vụ Đặt đơn và Thanh toán}

\begin{itemize}
    \item \textbf{Actor chính:} Khách hàng (\textit{Customer}). \textbf{Actor phụ:} Cổng thanh toán mô phỏng (\textit{Payment Gateway System}).
    \item \textbf{Liên kết yêu cầu:} \texttt{FR-10} đến \texttt{FR-17}, \texttt{FR-19}.
    \item \textbf{Tiền điều kiện:} Giỏ hàng có ít nhất 1 sản phẩm; khách hàng đã đăng nhập với JWT token hợp lệ.
    \item \textbf{Hậu điều kiện (Thành công):} Đơn hàng được tạo và chuyển trạng thái \texttt{PAID}; tồn kho sản phẩm giảm đúng số lượng đặt; lưu giao dịch thanh toán gắn \texttt{Idempotency-Key}; cộng điểm thưởng vào tài khoản; xóa giỏ hàng và hiển thị màn hình xác nhận đơn.
    \item \textbf{Hậu điều kiện (Thất bại):} Nếu chưa tạo đơn (lỗi tồn kho/chưa đăng nhập) thì không phát sinh đơn mới và giữ nguyên tồn kho. Nếu thanh toán thất bại (\texttt{PAYMENT\_FAILED}), đơn giữ trạng thái \texttt{PENDING}, giữ nguyên giỏ hàng và không cộng điểm thưởng để khách có thể thanh toán lại.
    \item \textbf{Luồng chính (Main Flow):}
    \begin{enumerate}
        \item Khách hàng nhấn ``Thanh toán'' tại màn hình giỏ hàng; hệ thống hiển thị danh sách món và tổng tạm tính (\texttt{UC09}).
        \item Hệ thống kiểm tra trạng thái đăng nhập (\texttt{UC08}).
        \item Khách hàng có thể nhập mã khuyến mãi; hệ thống kiểm tra tính hợp lệ và cập nhật tổng tiền sau giảm giá (\texttt{UC10}, \texttt{UC10a}).
        \item Khách hàng chọn phương thức thanh toán không tiền mặt (Ví/Thẻ/QR) và xác nhận (\texttt{UC11}).
        \item Hệ thống gọi \texttt{POST /orders} tạo đơn hàng trạng thái \texttt{PENDING}, kiểm tra và trừ tồn kho trong Database Transaction.
        \item Hệ thống sinh mã \texttt{Idempotency-Key} và gọi \texttt{POST /payments} gửi yêu cầu sang Cổng thanh toán.
        \item Cổng thanh toán trả về thành công; hệ thống cập nhật trạng thái đơn từ \texttt{PENDING} sang \texttt{PAID}.
        \item Hệ thống cộng điểm thưởng vào tài khoản khách hàng (\texttt{UC12}).
        \item Hệ thống xóa giỏ hàng và hiển thị màn hình xác nhận đơn hàng (\texttt{UC13}).
        \item Khách hàng theo dõi trạng thái xử lý đơn hàng (\texttt{UC17}).
    \end{enumerate}
    \item \textbf{Luồng thay thế (Alternative Flows):}
    \begin{itemize}
        \item \textbf{A1 -- Không dùng mã khuyến mãi:} Bỏ qua bước 3, giữ nguyên tổng tạm tính.
        \item \textbf{A2 -- Đổi mã khuyến mãi:} Nhập mã mới thay thế mã cũ hoặc xóa mã đã áp dụng (\texttt{UC10}).
        \item \textbf{A3 -- Gửi trùng yêu cầu thanh toán:} Khi nhận cùng \texttt{Idempotency-Key}, trả về kết quả giao dịch đầu tiên, không trừ tiền hay cộng điểm lần hai.
        \item \textbf{A4 -- Thanh toán lại đơn \texttt{PENDING}:} Khách có thể thanh toán lại trên đơn hàng chưa hoàn tất (\texttt{UC11a}).
    \end{itemize}
    \item \textbf{Luồng ngoại lệ (Exception Flows):}
    \begin{itemize}
        \item \textbf{E1 -- Chưa đăng nhập / hết phiên:} Chuyển hướng sang trang đăng nhập (\texttt{UC08}), sau khi đăng nhập quay lại trang thanh toán và giữ nguyên giỏ hàng.
        \item \textbf{E2 -- Giỏ hàng rỗng:} Chặn thao tác đặt đơn và thông báo giỏ hàng trống.
        \item \textbf{E3 -- Mã khuyến mãi không hợp lệ:} Thông báo lỗi, giữ nguyên tổng tiền và cho phép tiếp tục thanh toán.
        \item \textbf{E4 -- Không đủ tồn kho:} Rollback Transaction, thông báo món không đủ hàng, không tạo đơn và không thay đổi tồn kho.
        \item \textbf{E5 -- Thiếu \texttt{Idempotency-Key}:} Trả về mã lỗi \texttt{400 Bad Request}, từ chối tạo giao dịch.
        \item \textbf{E6 -- Đơn đã thanh toán:} Chặn thanh toán lại nếu đơn đã ở trạng thái \texttt{PAID}.
        \item \textbf{E7 -- Thanh toán thất bại (\texttt{PAYMENT\_FAILED}):} Giữ đơn ở trạng thái \texttt{PENDING}, giữ nguyên giỏ hàng, không cộng điểm và cấp khóa mới để thử lại.
        \item \textbf{E8 -- Cổng thanh toán không phản hồi:} Giữ đơn ở \texttt{PENDING}, không tự động tạo giao dịch trùng lặp.
    \end{itemize}
\end{itemize}

\begin{itemize}
    \item \textbf{Biến \texttt{DATABASE\_URL} (kết nối qua Connection Pooler -- PgBouncer):}
    \begin{verbatim}
DATABASE_URL="postgresql://postgres.pcpalhrohnzasnyfhklf:Khanggaming%4079@
aws-0-ap-northeast-2.pooler.supabase.com:5432/postgres?pgbouncer=true&connection_limit=1"

    Đường dẫn này được ứng dụng NestJS sử dụng xuyên suốt trong lúc chạy thực tế để thực hiện các truy vấn đọc thực đơn, tạo đơn hàng và chạy giao dịch thanh toán. Do mật khẩu quản trị cơ sở dữ liệu có chứa ký tự đặc biệt \texttt{@} (\texttt{Khanggaming@79}), nhóm đã chuyển ký tự này sang dạng mã hóa URL là \texttt{\%40} để tránh việc Prisma hiểu nhầm điểm bắt đầu của tên miền máy chủ. Đồng thời, hai tham số \texttt{pgbouncer=true} và \texttt{connection\_limit=1} được thêm vào cuối chuỗi nhằm giúp Prisma hoạt động ổn định qua lớp điều phối kết nối PgBouncer của Supabase mà không bị lỗi xung đột \textit{prepared statements}.

    \item \textbf{Biến \texttt{DIRECT\_URL} (kết nối trực tiếp):}
    \begin{verbatim}
DIRECT_URL="postgresql://postgres:Khanggaming%4079@
db.pcpalhrohnzasnyfhklf.supabase.co:5432/postgres"
    \end{verbatim}
    Đường dẫn này kết nối thẳng tới máy chủ PostgreSQL của dự án mà không đi qua bộ điều phối trung gian, được dùng riêng khi nhóm chạy lệnh đồng bộ lược đồ (\texttt{npx prisma db push} hoặc \texttt{npx prisma migrate deploy}) từ máy cục bộ lên đám mây.
\end{itemize}

Theo đúng nguyên tắc bảo mật đã thống nhất ở yêu cầu phi chức năng \texttt{NFR-08} (Mục 1.2.2), tệp \texttt{.env} được giữ lại ở máy cục bộ thông qua cấu hình \texttt{.gitignore}, còn các giá trị kết nối thực tế được nhập trực tiếp vào bảng điều khiển của nền tảng Cloud.

\subsubsection{Quy trình Deploy Frontend Next.js lên Vercel và cấu hình biến môi trường}
Sau khi tầng cơ sở dữ liệu trên Supabase và tầng Backend trên Render đã hoạt động ổn định, \textbf{Trần Thị Trúc Đào} thực hiện triển khai phần giao diện người dùng (Frontend Next.js) lên nền tảng \textbf{Vercel} tuần tự qua 4 bước:

\paragraph{Bước 1: Kiểm tra mã nguồn cục bộ và khởi tạo Project trên Vercel}
Trước khi đưa lên Cloud, nhóm chạy thử lệnh \texttt{npm run build} trong thư mục Frontend (\texttt{apps/web}) ở máy cá nhân để rà soát lỗi cú pháp TypeScript và đảm bảo toàn bộ 5 màn hình chính cùng 2 màn hình đăng nhập/đăng ký đều đóng gói thành công. Sau khi mã nguồn trên nhánh \texttt{main} của GitHub đã sẵn sàng, \textbf{Trần Thị Trúc Đào} truy cập trang chủ \texttt{https://vercel.com} và đăng nhập vào không gian làm việc nhóm (Team) có tên là \textbf{\texttt{djteam3}}, nhấn vào nút \textbf{Add New...} và chọn \textbf{Project} để tạo dự án mới.

\paragraph{Bước 2: Liên kết tài khoản GitHub và Import kho lưu trữ}
Tại màn hình khởi tạo dự án, thực hiện tích hợp Vercel với GitHub thông qua mục \textbf{Import Git Repository}, cấp quyền truy cập kho lưu trữ và chọn kho mã nguồn của đồ án là \textbf{\texttt{brew-lite}}, sau đó nhấn nút \textbf{Import} để chuyển sang bước cấu hình triển khai.

\paragraph{Bước 3: Cấu hình Framework và Biến môi trường \texttt{NEXT\_PUBLIC\_API\_URL}}
Ở màn hình cấu hình chi tiết của dự án (\textit{Configure Project}) và trang quản lý biến môi trường của Vercel, thiết lập các thông số kỹ thuật cốt lõi:
\begin{enumerate}
    \item \textbf{Chọn Framework (\textit{Framework Preset}):} Hệ thống tự động nhận diện và chọn đúng framework là \textbf{Next.js}.
    \item \textbf{Thư mục gốc và lệnh Build (\textit{Build and Output Settings}):} Trỏ thư mục gốc (\textit{Root Directory}) về thư mục Frontend (\texttt{apps/web}) trong Monorepo và giữ nguyên lệnh biên dịch mặc định của Next.js (\texttt{next build}).
    \item \textbf{Cấu hình chi tiết biến môi trường \texttt{NEXT\_PUBLIC\_API\_URL}:}
    Trong quá trình phát triển dưới máy cá nhân, Frontend gọi API thông qua địa chỉ cục bộ \texttt{http://localhost:3001}. Khi đưa lên Vercel, để trình duyệt của khách hàng kết nối được tới máy chủ Backend, \textbf{Trần Thị Trúc Đào} truy cập trang quản trị biến môi trường của dự án tại đường dẫn:
    \begin{center}
    \texttt{https://vercel.com/djteam3/brew-lite/settings/environment-variables}
    \end{center}
    Tại đây, tiến hành thêm mới biến môi trường \textbf{\texttt{NEXT\_PUBLIC\_API\_URL}} với quy tắc cấu hình như sau:
    \begin{itemize}
        \item \textbf{Tên biến (Key):} \texttt{NEXT\_PUBLIC\_API\_URL}. Tiền tố \textbf{\texttt{NEXT\_PUBLIC\_}} ở đầu tên biến là bắt buộc để trình biên dịch của Next.js nhúng giá trị đường dẫn này vào gói mã nguồn phía trình duyệt (Client-side bundle), giúp các màn hình chạy CSR (như chọn size/topping, giỏ hàng Zustand, thanh toán và đăng nhập) có thể gọi hàm \texttt{fetch(\`\$\{process.env.NEXT\_PUBLIC\_API\_URL\}/...\`)} trực tiếp tới Backend trên Render.
        \item \textbf{Giá trị (Value):} Điền đường dẫn HTTPS công khai của dịch vụ Backend NestJS đã được triển khai trên Render (thuộc dịch vụ \texttt{srv-dbo74m1srm7s73b580h0}, không có dấu gạch chéo \texttt{/} ở cuối đường dẫn).
        \item \textbf{Môi trường áp dụng (Environments):} Tích chọn cả 3 môi trường mặc định của Vercel gồm \textbf{Production} (bản chạy chính thức trên tên miền chính), \textbf{Preview} (bản chạy thử nghiệm cho các nhánh phụ) và \textbf{Development}, sau đó nhấn \textbf{Save}.
    \end{itemize}
\end{enumerate}

\paragraph{Bước 4: Cơ chế tự động cập nhật (CI/CD), quản lý tên miền và khôi phục phiên bản}
\begin{itemize}
    \item \textbf{Tự động triển khai khi có code mới (Auto-deploy Pipeline):} Mỗi khi một thành viên trong nhóm cập nhật mã nguồn và đẩy (\texttt{push}) lên nhánh \texttt{main} trên GitHub, Vercel tự động kích hoạt (trigger) đường ống pipeline, kéo bản code mới nhất về và tiến hành build bản mới mà không cần thao tác thủ công.
    \item \textbf{Lịch sử triển khai và khôi phục nhanh (\textit{Deployment History \& Rollback}):} Toàn bộ các bản triển khai trước đó được Vercel lưu lại trong mục \textit{Deployments} kèm nhật ký build (\textit{Build Logs}). Khi phát hiện sự cố sau triển khai, nhóm có thể dùng tính năng \textbf{Instant Rollback} trên Vercel để quay về phiên bản ổn định trước đó chỉ trong vài giây (đáp ứng yêu cầu \texttt{NFR-08} và \texttt{NFR-10}).
    \item \textbf{Quản lý đường dẫn và tên miền (\textit{Domains}):} Vercel cung cấp đường dẫn riêng cho từng lần deployment và một tên miền chính thức dạng \texttt{<tên-website>.vercel.app} luôn trỏ tới bản ổn định mới nhất trên nhánh \texttt{main}.
\end{itemize}

\subsubsection{Kiểm tra hoạt động thực tế sau khi triển khai lên Cloud}
Sau khi cả ba nền tảng \textbf{Vercel}, \textbf{Render} và \textbf{Supabase} đều báo trạng thái hoạt động xanh (Live/Ready), nhóm đã tiến hành kiểm tra thực tế toàn bộ luồng đặt đồ uống trên đường dẫn Cloud:
\begin{enumerate}
    \item \textbf{Luồng xem thực đơn và chọn món:} Trang chủ trên Vercel sử dụng biến \texttt{NEXT\_PUBLIC\_API\_URL} gọi thành công API \texttt{GET /products} từ Render và hiển thị đầy đủ danh sách món nước từ bảng \texttt{Product} trên Supabase; thao tác tìm kiếm, lọc danh mục, chọn Size/Topping và lưu giỏ hàng hoạt động chính xác.
    \item \textbf{Luồng đăng ký, đăng nhập và bảo vệ trang thanh toán:} Khi chuyển sang thanh toán mà chưa đăng nhập, trang web tự động điều hướng về \texttt{/login?redirect=/checkout}; sau khi đăng nhập, Backend trên Render xác thực mật khẩu băm (\texttt{bcrypt}) trong bảng \texttt{User} trên Supabase, cấp \texttt{access\_token} và đưa khách hàng quay lại trang thanh toán.
    \item \textbf{Luồng thanh toán không tiền mặt, khuyến mãi và xử lý ngoại lệ:} Khi áp dụng mã ưu đãi (\texttt{SGU2026}, \texttt{BREWLITE10}, \texttt{FREESHIP}) và nhấn xác nhận thanh toán, yêu cầu \texttt{POST /orders} và \texttt{POST /payments} kèm \texttt{Idempotency-Key} thực thi trọn vẹn trong Prisma Transaction trên Supabase: đơn chuyển từ \texttt{PENDING} sang \texttt{PAID}, tồn kho (\texttt{stock}) giảm đúng số lượng đặt và điểm tích lũy (\texttt{loyaltyPoints}) được cộng chính xác vào tài khoản. Khi bật chế độ giả lập lỗi cổng thanh toán, đơn giữ nguyên trạng thái chờ và bảo toàn giỏ hàng.
\end{enumerate}
